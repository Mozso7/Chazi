#include <stdio.h>
//Mind a 3 ciklussal megírni az első 100 pozitív egész szám összegét

void forMegoldas(){
    int osszeg = 0;
    for(int i = 1; i<= 100;i++){
        osszeg+= i;
    }
    printf("%d\n",osszeg);
}

void doMegoldas(){
    int i = 1;
    int osszeg = 0;
    while(i<=100){
      osszeg += i;
      i++;
    }
    printf("%d\n",osszeg);
}

void doWhileMegoldas(){
    int i = 1;
    int osszeg = 0;
    do{
        osszeg += i;
        i++;
    }while(i <= 100);
    printf("%d\n",osszeg);
    
    
}

int main() {
    forMegoldas();
    doMegoldas();
    doWhileMegoldas();

    return 0;
}
